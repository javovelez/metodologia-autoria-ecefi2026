% Aproximación trigonométrica del tren de pulsos rectangulares.
% T_0 = 4, omega_0 = pi/2, ventana centrada en t=0 con |t|<=1/2 y amplitud 1.
% a_0 = 1/4, a_k = (1/2) sinc(k*pi/4), b_k = 0.
% Valores: a_1=0.4502653, a_2=0.3183099, a_3=0.1500844, a_4=0,
%          a_5=-0.0900531, a_6=-0.1061033.
%
% Layout 6x2: filas k=1..6.
%   Columna izquierda: armónico individual a_k cos(k omega_0 t).
%   Columna derecha:   suma parcial S_k(t) sobre el pulso original.

\begin{document}
\begin{tikzpicture}[>={Latex[length=3pt,width=3pt]}, font=\scriptsize]

\begin{groupplot}[
    group style={
        group size=2 by 5,
        horizontal sep=1.2cm,
        vertical sep=0.45cm,
        x descriptions at=edge bottom,
    },
    width=5.5cm,
    height=3.6cm,
    xmin=-2.5, xmax=6.8,
    xtick={-2, 0, 2, 4, 6},
    enlarge x limits=0,
    every tick label/.style={font=\tiny},
    title style={
        font=\scriptsize,
        at={(axis description cs:1.0, 1.0)},
        anchor=north east,
        xshift=-3pt,
        yshift=-2pt,
    },
    xlabel={$t$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.18)}},
    axis lines=center,
    axis line style={->, thin, black},
    clip mode=individual,
]

% =============================================================
% Fila 1, k = 1
% =============================================================
% (a) armónico individual k=1
\nextgroupplot[
    ymin=-0.85, ymax=0.85,
    ytick={-0.5, 0, 0.5},
    title={(a)},
]
    \addplot[red!80!black, semithick, samples=200, domain=-2.3:6.3, smooth]
        {0.4502653*cos(deg(pi/2*x))};

% (b) suma parcial S_1(t) sobre el pulso
\nextgroupplot[
    ymin=-0.3, ymax=1.4,
    ytick={0, 0.5, 1},
    yticklabel style={xshift=-5pt},
    title={(b)},
]
    \addplot[utnblue, semithick]
        coordinates {
            (-2.3, 0) (-0.5, 0) (-0.5, 1) (0.5, 1) (0.5, 0)
            (3.5, 0) (3.5, 1) (4.5, 1) (4.5, 0) (6.3, 0)
        };
    \addplot[red!80!black, semithick, samples=300, domain=-2.3:6.3, smooth]
        {0.25 + 0.4502653*cos(deg(pi/2*x))};

% =============================================================
% Fila 2, k = 2
% =============================================================
% (c) armónico individual k=2
\nextgroupplot[
    ymin=-0.85, ymax=0.85,

    title={(c)},
]

        {0.3183099*cos(deg(pi*x))};

% (d) suma parcial S_2(t)
\nextgroupplot[
    ymin=-0.3, ymax=1.4,
    ytick={0, 0.5, 1},
    yticklabel style={xshift=-5pt},
    title={(d)},
]
    \addplot[utnblue, semithick]

        };
    \addplot[red!80!black, semithick, samples=400, domain=-2.3:6.3, smooth]
        {0.25
         + 0.4502653*cos(deg(pi/2*x))
         + 0.3183099*cos(deg(pi*x))};

% =============================================================
% Fila 3, k = 3

    title={(e)},
