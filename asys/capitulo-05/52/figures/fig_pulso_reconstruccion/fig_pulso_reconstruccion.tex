% Tren periódico de pulsos rectangulares con tau = T_0/4, T_0 = 1.
% Coeficientes c_k = (1/4) sinc(k pi / 4); ceros en k = +/- 4, +/- 8, +/- 12 ...
% Reconstrucción truncada hat_x_N(t) = sum_{k=-N}^{N} c_k e^{j k w_0 t}
%                                    = c_0 + 2 sum_{k=1}^{N} c_k cos(k w_0 t)
% Cuatro paneles apilados con N = 1, 3, 7, 21. Se grafican tres períodos
% completos (-1.5 T_0 a 1.5 T_0) para mostrar la naturaleza periódica del tren.

% Macro: construye la expresión de la suma parcial hasta N harmónicos.
% Salta los k múltiplos de 4 (donde c_k = 0).
\newcommand{\BuildPartialSum}[1]{%
    \xdef\PartialSumExpr{0.25}%
    \foreach \k in {1,...,#1} {%
        \pgfmathtruncatemacro{\rem}{mod(\k,4)}%
        \ifnum\rem>0
            \xdef\PartialSumExpr{\PartialSumExpr
                + 2*sin(deg(\k*pi/4))/(\k*pi)*cos(deg(2*pi*\k*x))}%
        \fi
    }%
}
