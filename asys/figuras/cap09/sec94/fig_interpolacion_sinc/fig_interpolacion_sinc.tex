% fig_interpolacion_sinc
% Delta t = 1. Muestras en n=-1..4 con valores v_n. Cada muestra siembra
% v_n * sinc(pi(t-n)) = v_n*sin(180(x-n))/(pi(x-n)). La suma reconstruye x(t).
% Dominio elegido para no caer exactamente sobre enteros (evita 0/0).

\begin{document}
\begin{tikzpicture}[>={Latex[length=3pt]}, font=\scriptsize]
\begin{axis}[
    width=13.5cm, height=6.2cm,
    axis lines=center,
    axis line style={->, thick},
    xmin=-1.7, xmax=4.9, ymin=-0.55, ymax=1.28,
    xtick={-1,0,1,2,3,4},
    xticklabels={$-\Delta t$,$0$,$\Delta t$,$2\Delta t$,$3\Delta t$,$4\Delta t$},
    ytick={0,1},
    every tick label/.style={font=\tiny},
    xlabel={$t$}, xlabel style={anchor=west, at={(axis description cs:1.0,0.46)}},
    enlarge x limits=false, clip mode=individual, axis on top,
    samples=320, domain=-1.6:4.8,
]
    % senos cardinales individuales (finos, colores)
    \addplot[utnblue!55, thin] {0.20*sin(180*(x+1))/(pi*(x+1))};
    \addplot[red!55!black, thin] {0.65*sin(180*(x))/(pi*(x))};
    \addplot[codegreen!70!black, thin] {1.00*sin(180*(x-1))/(pi*(x-1))};
    \addplot[codepurple!70, thin] {0.75*sin(180*(x-2))/(pi*(x-2))};
    \addplot[orange!80!black, thin] {0.20*sin(180*(x-3))/(pi*(x-3))};
    \addplot[teal!70!black, thin] {-0.25*sin(180*(x-4))/(pi*(x-4))};

    % suma = reconstruccion
    \addplot[utnblue, very thick] {
        0.20*sin(180*(x+1))/(pi*(x+1))
        + 0.65*sin(180*(x))/(pi*(x))
        + 1.00*sin(180*(x-1))/(pi*(x-1))
        + 0.75*sin(180*(x-2))/(pi*(x-2))
        + 0.20*sin(180*(x-3))/(pi*(x-3))
        - 0.25*sin(180*(x-4))/(pi*(x-4))
    };

    % muestras
    \addplot[only marks, mark=*, mark size=1.8pt,
             mark options={fill=black, draw=black}]
        coordinates {(-1,0.20) (0,0.65) (1,1.00) (2,0.75) (3,0.20) (4,-0.25)};
\end{axis}
\end{tikzpicture}
\end{document}
