% fig_ejemplo_nyquist
% Ejemplo de 9.3.1: x(t) = cos(4 pi t) + cos(10 pi t)
%   componentes: 2 ciclos/s -> 4 pi rad/s ; 5 ciclos/s -> 10 pi rad/s
%   wM = 10 pi ; velocidad de Nyquist 2wM = 20 pi
%   muestreo elegido: 12 muestras/s -> ws = 24 pi ; frecuencia de Nyquist ws/2 = 12 pi
%   franja libre entre wM = 10 pi y ws - wM = 14 pi (ancho 4 pi)
% Unidad del eje: 1 unidad de dibujo = 2 pi rad/s.
% (a) espectro de x(t): cuatro lineas, la mas alta en +-wM.
% (b) espectro muestreado: copia central + copias en +-ws; banda base sombreada
%     y frecuencia de Nyquist +-ws/2 en linea gris discontinua.

\begin{document}
\begin{tikzpicture}[>={Latex[length=4pt]}, font=\scriptsize, xscale=0.32,
    lin/.style={utnblue, thick, -{Latex[length=4pt]}},
    linf/.style={utnblue!40, thick, -{Latex[length=3.5pt]}},
    nyq/.style={black!45, thin, dashed},
    ax/.style={->, thin, black},
    tick/.style={thin},
]
    \def\H{0.9}
    \def\Hs{1.20}

    % ===================== (a) espectro de x(t) =====================
    \begin{scope}[shift={(0,0)}]
        \node at (-18.6,0.7) {(a)};
        \draw[ax] (-18,0) -- (18.6,0) node[right] {$\omega$};
        \draw[ax] (0,0) -- (0,\H+0.30);
        \node[anchor=north east, font=\tiny, inner sep=1.5pt] at (-0.15,-0.02) {$0$};
        \draw[tick] ([xshift=-2pt]0,\H) -- ([xshift=2pt]0,\H);
        \node[anchor=east, font=\tiny, inner sep=2.5pt] at ([xshift=-2pt]0,\H) {$\pi$};
        % lineas del espectro
        \foreach \x in {-5,-2,2,5} { \draw[lin] (\x,0) -- (\x,\H); }
        % marcas del eje
        \foreach \x/\lab in {2/{4\pi}, 5/{\omega_M=10\pi}, -2/{-4\pi}, -5/{-10\pi}} {
            \draw[tick] (\x,0.05) -- (\x,-0.05) node[below, font=\tiny] {$\lab$};
        }
    \end{scope}

    % ===================== (b) espectro muestreado =====================
    \begin{scope}[shift={(0,-2.9)}]
        \node at (-18.6,0.7) {(b)};
        % banda base
        \fill[gray!10] (-6,0) rectangle (6,\Hs+0.10);
        \node[black!60, font=\tiny, anchor=south] at (-3.0,\Hs+0.12) {banda base};
        \draw[ax] (-18,0) -- (18.6,0) node[right] {$\omega$};
        \draw[ax] (0,0) -- (0,\Hs+0.30);
        \node[anchor=north east, font=\tiny, inner sep=1.5pt] at (-0.15,-0.02) {$0$};
        % copias vecinas, centradas en +-ws
        \foreach \x in {7,10,14,17,-7,-10,-14,-17} { \draw[linf] (\x,0) -- (\x,\Hs); }
        % copia central
        \foreach \x in {-5,-2,2,5} { \draw[lin] (\x,0) -- (\x,\Hs); }
        % peso de cada impulso, escalado por 1/Delta t
        \draw[tick] ([xshift=-2pt]0,\Hs) -- ([xshift=2pt]0,\Hs);
        \node[anchor=east, font=\tiny, inner sep=3pt] at ([xshift=-1pt]0,\Hs)
            {$\tfrac{\pi}{\Delta t}$};
        % frecuencia de Nyquist
        \draw[nyq] (6,0) -- (6,\Hs+0.22);
        \draw[nyq] (-6,0) -- (-6,\Hs+0.22);
        \node[black!60, anchor=south, font=\tiny] at (6,\Hs+0.22) {$\tfrac{\omega_s}{2}=12\pi$};
        \node[black!60, anchor=south, font=\tiny] at (-6,\Hs+0.22) {$-\tfrac{\omega_s}{2}$};
        % franja libre entre wM y ws - wM
        \draw[<->, black!55] (5,0.34) -- (7,0.34);
        \node[black!60, anchor=south, font=\tiny, fill=white, inner sep=1pt]
            at (6.0,0.36) {franja libre};
        % marcas del eje
        \foreach \x/\lab in {5/{\omega_M}, 12/{\omega_s=24\pi}, -12/{-24\pi}} {
            \draw[tick] (\x,0.05) -- (\x,-0.05) node[below, font=\tiny] {$\lab$};
        }
        \draw[tick] (7,0.05) -- (7,-0.05);
        \node[anchor=north, font=\tiny] at (7,-0.45) {$\omega_s\!-\!\omega_M\!=\!14\pi$};
        \draw[thin, -{Latex[length=3pt]}] (7,-0.45) -- (7,-0.13);
    \end{scope}
\end{tikzpicture}
\end{document}
