    panel/.style={

            font=\small, rotate=0, anchor=south east,

% ---- panel (a): x(t), triangulo en [0,2] pico en t=1 ----
\begin{axis}[panel, name=ax1,
    ylabel={$x(t)$},
    title={(a)}, title style={font=\small\bfseries}]
\addplot[utnblue, very thick] coordinates {(-1.5,0)(0,0)(1,1)(2,0)(2.5,0)};
\end{axis}

% ---- panel (b): x(-2t), triangulo en [-1,0] pico en t=-1/2 ----
\begin{axis}[panel, name=ax2,
    at={(ax1.below south west)}, anchor=above north west, yshift=-6pt,
    ylabel={$x(-2t)$},
    title={(b)}, title style={font=\small\bfseries}]
\addplot[red!80!black, very thick] coordinates {(-1.5,0)(-1,0)(-0.5,1)(0,0)(2.5,0)};
\node[above, font=\scriptsize, gray] at (axis cs:-0.5,1.1) {pico en $-\tfrac{1}{2}$};
\end{axis}

% ---- panel (c): x(-2t+3), triangulo en [1/2, 3/2] pico en t=1 ----
\begin{axis}[panel, name=ax3,

    ylabel={$x(-2t{+}3)$},
    title={(c)}, title style={font=\small\bfseries}]
\addplot[red!80!black, very thick] coordinates {(-1.5,0)(0.5,0)(1,1)(1.5,0)(2.5,0)};
\node[above, font=\scriptsize, gray] at (axis cs:1,1.1) {pico en $1$};
\end{axis}
