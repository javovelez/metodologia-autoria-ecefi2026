% Base armónica trigonométrica para señales periódicas reales.
% Se grafican las primeras tres parejas {cos(k w_0 t), sin(k w_0 t)}
% para k = 1, 2, 3 sobre un período.
% T_0 = 1, w_0 = 2 pi.

\begin{document}
\begin{tikzpicture}[>={Latex[length=3pt,width=3pt]}, font=\footnotesize]

\begin{groupplot}[
    group style={
        group size=3 by 2,
        horizontal sep=1.0cm,
        vertical sep=0.95cm,
        x descriptions at=edge bottom,
        y descriptions at=edge left,
    },
    width=5.5cm,
    height=3.6cm,
    xmin=-0.04, xmax=1.10,
    ymin=-1.4, ymax=1.4,
    axis lines=center,
    axis line style={->, thin, black},
    xtick={0, 0.5, 1},
    xticklabels={$0$, $T_0/2$, $T_0$},
    ytick={-1, 0, 1},
    enlarge x limits=0,
    every tick label/.style={font=\tiny},
    title style={
        font=\scriptsize,
        at={(axis description cs:1.0, 1.0)},
        anchor=north east,
        xshift=-3pt,
        yshift=-2pt,
    },
    clip mode=individual,
]

% Fila superior: cosenos
\nextgroupplot[title={(a)}]
    \addplot[utnblue, semithick, domain=0:1.05, samples=200, smooth]
        {cos(deg(2*pi*x))};

\nextgroupplot[title={(b)}]
    \addplot[utnblue, semithick, domain=0:1.05, samples=300, smooth]
        {cos(deg(4*pi*x))};

\nextgroupplot[title={(c)}]
    \addplot[utnblue, semithick, domain=0:1.05, samples=400, smooth]
        {cos(deg(6*pi*x))};

% Fila inferior: senos
\nextgroupplot[title={(d)}]
    \addplot[red!80!black, semithick, domain=0:1.05, samples=200, smooth]
        {sin(deg(2*pi*x))};

\nextgroupplot[title={(e)}]
    \addplot[red!80!black, semithick, domain=0:1.05, samples=300, smooth]
        {sin(deg(4*pi*x))};

\nextgroupplot[title={(f)}]
    \addplot[red!80!black, semithick, domain=0:1.05, samples=400, smooth]
        {sin(deg(6*pi*x))};

\end{groupplot}

\end{tikzpicture}
\end{document}
