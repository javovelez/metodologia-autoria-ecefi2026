\begin{tikzpicture}[>=Latex, scale=1.0, font=\small]

    % =====================
    % PANEL A: Serie Convergente
    % =====================
    \begin{scope}[scale=2]
        % Título
        \node[font=\bfseries] at (0.85, 2.08) {(a) Convergente ($|z_1|<1$)};
        
        % Ejes
        \draw[->, thick, black] (-0.1, 0) -- (1.7, 0) node[right] {Re};
        \draw[->, thick, black] (0, -0.1) -- (0, 1.7) node[above] {Im};
        
        % z = 0.75 * e^{j*50°}
        \def\r{0.75}
        \def\ang{50}
        
        % Coordenadas de los términos
        \coordinate (P0) at (0, 0);
        \coordinate (P1) at (1, 0);
        
        \pgfmathsetmacro{\zRe}{\r*cos(\ang)}
        \pgfmathsetmacro{\zIm}{\r*sin(\ang)}
        \coordinate (P2) at ({1 + \zRe}, {\zIm});
        
        \pgfmathsetmacro{\zzRe}{\r*\r*cos(2*\ang)}
        \pgfmathsetmacro{\zzIm}{\r*\r*sin(2*\ang)}
        \coordinate (P3) at ({1 + \zRe + \zzRe}, {\zIm + \zzIm});
        
        \pgfmathsetmacro{\zzzRe}{\r*\r*\r*cos(3*\ang)}
        \pgfmathsetmacro{\zzzIm}{\r*\r*\r*sin(3*\ang)}
        \coordinate (P4) at ({1 + \zRe + \zzRe + \zzzRe}, {\zIm + \zzIm + \zzzIm});
        
        \pgfmathsetmacro{\zzzzRe}{\r*\r*\r*\r*cos(4*\ang)}
        \pgfmathsetmacro{\zzzzIm}{\r*\r*\r*\r*sin(4*\ang)}
        \coordinate (P5) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm});
        
        \pgfmathsetmacro{\zfiveRe}{\r*\r*\r*\r*\r*cos(5*\ang)}
        \pgfmathsetmacro{\zfiveIm}{\r*\r*\r*\r*\r*sin(5*\ang)}
        \coordinate (P6) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm});
        
        \pgfmathsetmacro{\zsixRe}{\r*\r*\r*\r*\r*\r*cos(6*\ang)}
        \pgfmathsetmacro{\zsixIm}{\r*\r*\r*\r*\r*\r*sin(6*\ang)}
        \coordinate (P7) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm});
        
        \pgfmathsetmacro{\zsevenRe}{\r*\r*\r*\r*\r*\r*\r*cos(7*\ang)}
        \pgfmathsetmacro{\zsevenIm}{\r*\r*\r*\r*\r*\r*\r*sin(7*\ang)}
        \coordinate (P8) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe + \zsevenRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm + \zsevenIm});

        \pgfmathsetmacro{\zeightRe}{\r*\r*\r*\r*\r*\r*\r*\r*cos(8*\ang)}
        \pgfmathsetmacro{\zeightIm}{\r*\r*\r*\r*\r*\r*\r*\r*sin(8*\ang)}
        \coordinate (P9) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe + \zsevenRe + \zeightRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm + \zsevenIm + \zeightIm});

        \pgfmathsetmacro{\znineRe}{\r*\r*\r*\r*\r*\r*\r*\r*\r*cos(9*\ang)}
        \pgfmathsetmacro{\znineIm}{\r*\r*\r*\r*\r*\r*\r*\r*\r*sin(9*\ang)}
        \coordinate (P10) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe + \zsevenRe + \zeightRe + \znineRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm + \zsevenIm + \zeightIm + \znineIm});

        \pgfmathsetmacro{\ztenRe}{\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*cos(10*\ang)}
        \pgfmathsetmacro{\ztenIm}{\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*sin(10*\ang)}
        \coordinate (P11) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe + \zsevenRe + \zeightRe + \znineRe + \ztenRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm + \zsevenIm + \zeightIm + \znineIm + \ztenIm});

        \pgfmathsetmacro{\zelevenRe}{\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*cos(11*\ang)}
        \pgfmathsetmacro{\zelevenIm}{\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*\r*sin(11*\ang)}
        \coordinate (P12) at ({1 + \zRe + \zzRe + \zzzRe + \zzzzRe + \zfiveRe + \zsixRe + \zsevenRe + \zeightRe + \znineRe + \ztenRe + \zelevenRe}, {\zIm + \zzIm + \zzzIm + \zzzzIm + \zfiveIm + \zsixIm + \zsevenIm + \zeightIm + \znineIm + \ztenIm + \zelevenIm});

        % Límite S_inf = 1/(1-z)
        \pgfmathsetmacro{\denomRe}{1 - \r*cos(\ang)}
        \pgfmathsetmacro{\denomIm}{-\r*sin(\ang)}
        \pgfmathsetmacro{\denomMag}{\denomRe*\denomRe + \denomIm*\denomIm}
        \pgfmathsetmacro{\SinfRe}{\denomRe/\denomMag}
        \pgfmathsetmacro{\SinfIm}{-\denomIm/\denomMag}
        \coordinate (Sinf) at (\SinfRe, \SinfIm);
        
        % Vectores con flechas decrecientes (hasta término 8, luego constante)
        \draw[-{Latex[length=3mm, width=2mm]}, very thick, utnblue] (P0) -- (P1);
        \draw[-{Latex[length=2.6mm, width=1.7mm]}, thick, utnblue] (P1) -- (P2);
        \draw[-{Latex[length=2.2mm, width=1.4mm]}, thick, utnblue] (P2) -- (P3);
        \draw[-{Latex[length=1.8mm, width=1.2mm]}, semithick, utnblue] (P3) -- (P4);
        \draw[-{Latex[length=1.4mm, width=1.0mm]}, semithick, utnblue] (P4) -- (P5);
        \draw[-{Latex[length=1.1mm, width=0.8mm]}, thin, utnblue] (P5) -- (P6);
        \draw[-{Latex[length=0.8mm, width=0.6mm]}, thin, utnblue] (P6) -- (P7);
        \draw[-{Latex[length=0.6mm, width=0.4mm]}, thin, utnblue] (P7) -- (P8);
        % A partir del término 8, tamaño constante
        \draw[-{Latex[length=0.6mm, width=0.4mm]}, thin, utnblue] (P8) -- (P9);
        \draw[-{Latex[length=0.6mm, width=0.4mm]}, thin, utnblue] (P9) -- (P10);
        \draw[-{Latex[length=0.6mm, width=0.4mm]}, thin, utnblue] (P10) -- (P11);
        \draw[-{Latex[length=0.6mm, width=0.4mm]}, thin, utnblue] (P11) -- (P12);

        % Punto límite (en naranja)
        \fill[orange] (Sinf) circle (1pt);
        \node[above, font=\footnotesize, orange] at (Sinf) {$\frac{1}{1-z_1}$};
        
        % Etiquetas de sumas parciales en las puntas de los vectores
        \node[below, font=\scriptsize] at (P1) {$z_1^0$};
        \node[right, font=\scriptsize] at (P2) {$z_1^0+z_1^1$};
        \node[right, font=\scriptsize] at (P3) {$z_1^0+z_1^1+z_1^2$};
        \node[above, font=\scriptsize] at (P4) {$z_1^0+z_1^1+z_1^2+z_1^3$};
        \node[left, font=\scriptsize] at (P5) {$z_1^0+z_1^1+z_1^2+z_1^3+z_1^4$};
        \node[left, font=\scriptsize] at (P6) {$z_1^0+z_1^1+\cdots+z_1^5$};
    \end{scope}
    
    % =====================
    % PANEL B: Serie Divergente
    % =====================
    \begin{scope}[shift={(5, 0)}, scale=1.0]
        % Título
        \node[font=\bfseries] at (2.1, 4.1) {(b) Divergente ($|z_2|>1$)};
        
        % Ejes
        \draw[->, thick, black] (-0.1, 0) -- (3.5, 0) node[right] {Re};
        \draw[->, thick, black] (0, -0.1) -- (0, 3.5) node[above] {Im};
        
        % z = 1.12 * e^{j*25°}
        \def\rd{1.12}
        \def\angd{25}
        
        % Coordenadas
        \coordinate (D0) at (0, 0);
        \coordinate (D1) at (1, 0);
        
        \pgfmathsetmacro{\dzRe}{\rd*cos(\angd)}
        \pgfmathsetmacro{\dzIm}{\rd*sin(\angd)}
        \coordinate (D2) at ({1 + \dzRe}, {\dzIm});
        
        \pgfmathsetmacro{\dzzRe}{\rd*\rd*cos(2*\angd)}
        \pgfmathsetmacro{\dzzIm}{\rd*\rd*sin(2*\angd)}
        \coordinate (D3) at ({1 + \dzRe + \dzzRe}, {\dzIm + \dzzIm});
        
        \pgfmathsetmacro{\dzzzRe}{\rd*\rd*\rd*cos(3*\angd)}
        \pgfmathsetmacro{\dzzzIm}{\rd*\rd*\rd*sin(3*\angd)}
        \coordinate (D4) at ({1 + \dzRe + \dzzRe + \dzzzRe}, {\dzIm + \dzzIm + \dzzzIm});

        \pgfmathsetmacro{\dzzzzRe}{\rd*\rd*\rd*\rd*cos(4*\angd)}
        \pgfmathsetmacro{\dzzzzIm}{\rd*\rd*\rd*\rd*sin(4*\angd)}
        \coordinate (D5) at ({1 + \dzRe + \dzzRe + \dzzzRe + \dzzzzRe}, {\dzIm + \dzzIm + \dzzzIm + \dzzzzIm});
        
        % Punto intermedio para el último vector (mitad)
        \coordinate (D5half) at ($(D4)!0.4!(D5)$);
        
        % Vectores divergentes (mismo tamaño de flecha y grosor)
        \draw[-{Latex[length=2mm, width=1.3mm]}, thick, red!80!black] (D0) -- (D1);
        \draw[-{Latex[length=2mm, width=1.3mm]}, thick, red!80!black] (D1) -- (D2);
        \draw[-{Latex[length=2mm, width=1.3mm]}, thick, red!80!black] (D2) -- (D3);
        \draw[-{Latex[length=2mm, width=1.3mm]}, thick, red!80!black] (D3) -- (D4);
        % Último vector a la mitad y discontinuo (sin punta)
        \draw[thick, red!80!black, dashed] (D4) -- (D5half);
        
        % Indicador de divergencia en otro color (naranja)
        \node[orange, font=\bfseries] at ($(D5half) + (0.6, -0.3)$) {$\to \infty$};
        
        % Etiquetas de sumas parciales
        \node[below, font=\scriptsize] at (D1) {$z_2^0$};
        \node[right, font=\scriptsize] at (D2) {$z_2^0+z_2^1$};
        \node[left, font=\scriptsize] at (D3) {$z_2^0+z_2^1+z_2^2$};
        \node[left, font=\scriptsize] at (D4) {$z_2^0+z_2^1+z_2^2+z_2^3$};
    \end{scope}
    
    % =====================
    % PANEL C: Región de Convergencia (reducido)
    % =====================
    \begin{scope}[shift={(12, 1.5)}, scale=0.75]
        % Título
        \node[font=\bfseries] at (0.1, 3.5) {(c) Región de Convergencia};
        
        % Disco unitario (sombreado) - PRIMERO
        \fill[cyan!15] (0, 0) circle (2.2);

        \draw[thick] (0.08, -2.2) -- (-0.08, -2.2) node[left, font=\footnotesize] {$-j$};
        
        % Punto z1 (convergente, dentro del círculo)
        \pgfmathsetmacro{\zoneRe}{0.75*cos(50)*2.2}
        \pgfmathsetmacro{\zoneIm}{0.75*sin(50)*2.2}
        \fill[utnblue] (\zoneRe, \zoneIm) circle (2.5pt);
        \node[above , font=\footnotesize, utnblue] at (\zoneRe, \zoneIm) {$z_1$};
        
        % Punto z2 (divergente, fuera del círculo)
        \pgfmathsetmacro{\ztwoRe}{1.12*cos(25)*2.2}
        \pgfmathsetmacro{\ztwoIm}{1.12*sin(25)*2.2}
        \fill[red!80!black] (\ztwoRe, \ztwoIm) circle (2.5pt);
        \node[above , font=\footnotesize, red!80!black] at (\ztwoRe, \ztwoIm) {$z_2$};
        
        % Etiqueta ROC
        \node at (0.7, 0.5) {$|z| < 1$};

    \end{scope}
