\begin{tikzpicture}[>=Latex, scale=1.2, font=\small,
    arrowblue/.style={decoration={markings, mark=at position 0.55 with {\arrow[utnblue, scale=1.3]{Latex}}}, postaction={decorate}},
    arroworange/.style={decoration={markings, mark=at position 0.55 with {\arrow[orange, scale=1.3]{Latex}}}, postaction={decorate}},
    arrowred/.style={decoration={markings, mark=at position 0.5 with {\arrow[red, scale=1.3]{Latex}}}, postaction={decorate}}
]

    % Ejes coordenados
    \draw[->, gray!70, line width=0.8pt] (-0.5, 0) -- (6, 0) node[right] {$\text{Re}$};
    \draw[->, gray!70, line width=0.8pt] (0, -0.5) -- (0, 3.5) node[above] {$\text{Im}$};

    % Puntos clave (suaves, sin recubrimiento brusco)

    \coordinate (B) at (3.5, 2.5);
    \coordinate (C) at (5, 1);

    % Camino C1 (Azul)
    \draw[utnblue, line width=1.5pt] (A) .. controls (1.5, 2.5) and (2.5, 2.5) .. (B);
    \path[arrowblue] (A) .. controls (1.5, 2.5) and (2.5, 2.5) .. (B);
    \node[utnblue, above left] at (1.8, 1.9) {$C_1$};

    % Camino C2 (Naranja)
    \draw[orange, line width=1.5pt] (B) .. controls (4.5, 2.5) and (4.5, 1.5) .. (C);
    \path[arroworange] (B) .. controls (4.5, 2.5) and (4.5, 1.5) .. (C);
    \node[orange, above right] at (4.3, 2.2) {$C_2$};

    % Nodos intermedios con viñetas limpias
    \filldraw[gray!80!black] (A) circle (1.8pt);
    \node[below left, gray!80!black] at (A) {$z_A$};
    
    \filldraw[gray!80!black] (B) circle (1.8pt);
    \node[above, gray!80!black, yshift=2pt] at (B) {$z_{int}$};
    
    \filldraw[gray!80!black] (C) circle (1.8pt);
    \node[below right, gray!80!black] at (C) {$z_B$};
    
    % Etiqueta del camino general C
    \node[gray!90!black, font=\bfseries] at (2.5, 3) {Camino Total $C = C_1 \cup C_2$};
