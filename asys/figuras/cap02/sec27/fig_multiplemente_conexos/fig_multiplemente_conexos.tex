        \coordinate (Z1) at (135:1.4);

        \coordinate (Z1_out_outer) at (135+\aOut:\R);

        \coordinate (Z1_out_inner) at ($ (Z1) + (-225+\aIn:\r) $); % 135+aIn - 360

        \coordinate (Z2_out_outer) at (315+\aOut:\R);

        \coordinate (Z2_out_inner) at ($ (Z2) + (-45+\aIn:\r) $); % 315+aIn - 360

        \node at (0, -\R-0.8) {\textbf{(b) Cortes y Rodeo}};
    \end{scope}
