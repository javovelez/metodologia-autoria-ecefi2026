\begin{document}
    \begin{tikzpicture}[>=Latex, scale=0.75, font=\small]
        % --- PLANO Z (Primer Cuadrante) ---
        % Zoom x2: scale=2 localmente. Ajustamos ejes para mantener tamaño visual.
        \begin{scope}[scale=2]
            % Rayos (Azul -> Cian)
            \foreach \a in {0, 30, 60, 90} {
                \pgfmathsetmacro{\per}{\a/90 * 100}
                \draw[line width=0.8pt, color=blue!\per!cyan] (0,0) -- (\a:2.2);
            }
            
            % Círculos (Rojo -> Amarillo)
            \foreach \r in {0.5, 1, 1.5, 2} {
                \pgfmathsetmacro{\per}{(\r)/2 * 100}
                \draw[line width=0.8pt, color=red!\per!yellow] (\r,0) arc (0:90:\r);
            }
            
            % Ejes (dibujados al final para quedar encima)
            \draw[->] (-0.25,0) -- (2.25,0) node[right] {$x$};
            \draw[->] (0,-0.25) -- (0,2.25) node[above] {$y$};
            \node at (1, -0.5) {\textbf{Plano Z}};
            
            % Etiquetas en eje x
            \foreach \x in {0.5, 1, 1.5, 2} {
                \draw (\x, 0.05) -- (\x, -0.05) node[below, font=\tiny] {\x};
            }
        \end{scope}
        
        % Flecha
        \draw[->, thick, bend left] (5, 2.5) to node[midway, above] {$w=z^2$} (7, 2.5);
        
        % --- PLANO W (Dos Primeros Cuadrantes) ---
        \begin{scope}[shift={(12,0)}]
            % Rayos Transformados (Azul -> Cian) - theta' = 2*theta
            \foreach \a in {0, 30, 60, 90} {
                \pgfmathsetmacro{\per}{\a/90 * 100}
                \draw[line width=0.8pt, color=blue!\per!cyan] (0,0) -- ({2*\a}:4.2);
            }
            
            % Círculos Transformados (Rojo -> Amarillo) - r' = r^2
            \foreach \r in {0.5, 1, 1.5, 2} {
                \pgfmathsetmacro{\per}{(\r)/2 * 100}
                \draw[line width=0.8pt, color=red!\per!yellow] ({\r*\r},0) arc (0:180:{\r*\r});
            }
            
            % Ejes (dibujados al final)
            \draw[->] (-4.5,0) -- (4.5,0) node[right] {$u$};
            \draw[->] (0,-0.5) -- (0,5) node[above] {$v$};
            \node at (0, -1) {\textbf{Plano W}};
            
            % Etiquetas en eje u (radios transformados)
            \foreach \val in {0.25, 1, 2.25, 4} {
                \draw (\val, 0.05) -- (\val, -0.05) node[below, font=\tiny] {\val};
            }
        \end{scope}
    \end{tikzpicture}
\end{document}
