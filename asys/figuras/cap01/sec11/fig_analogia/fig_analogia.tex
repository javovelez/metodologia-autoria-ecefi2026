        \draw[->, ultra thick, red!70!black] (2, 2.2) -- (2, 1) node[right, font=\footnotesize] {$f(t)$};

        \node[font=\footnotesize, anchor=north] at (2, -1.2) {

            Posición (salida): & $\textcolor{codegreen!70!black}{y(t)}$
            \end{tabular}

            \end{tabular}
