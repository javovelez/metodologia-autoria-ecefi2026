% fig_masa_resorte_libre.tex
% Respuesta libre del oscilador masa-resorte de §7.5.5:
%   y(t) = e^{-t} [ cos(3t) + (1/3) sin(3t) ],  y(0) = 1, y'(0) = 0.
% Sinusoide amortiguada con envolvente exponencial
%   ±sqrt(1 + 1/9) e^{-t} ≈ ±1.054 e^{-t} (a trazos).
% Dibujada con factor vertical 1.5 (sketch a escala uniforme).
% ============================================================
%\PassOptionsToPackage{gray}{xcolor}

\documentclass[border=5mm]{standalone}
\input{../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=0.85, font=\small]

% --- Ejes ---
\draw[->, thick] (-0.3, 0) -- (4.8, 0) node[right, font=\footnotesize] {$t$};
\draw[->, thick] (0, -1.4) -- (0, 1.9) node[right, font=\footnotesize] {$y(t)$};

% Marca del valor inicial y(0) = 1 (1.5 en coordenadas de dibujo)
\draw (-0.07, 1.5) -- (0.07, 1.5);
\node[left, font=\scriptsize] at (-0.09, 1.5) {$1$};

% --- Envolvente exponencial (a trazos) ---
\draw[gray, thick, dashed, smooth, samples=60, domain=0:4.4]
    plot (\x, {1.5*1.054*exp(-\x)});
\draw[gray, thick, dashed, smooth, samples=60, domain=0:4.4]
    plot (\x, {-1.5*1.054*exp(-\x)});

% --- Respuesta libre ---
\draw[utnblue!90!black, very thick, smooth, samples=300, domain=0:4.4]
    plot (\x, {1.5*exp(-\x)*(cos(deg(3*\x)) + sin(deg(3*\x))/3)});

\end{tikzpicture}
\end{document}
