% atraviesa un transitorio (oscilación amortiguada, banda cyan!15) y
% se asienta en el nivel permanente y_infty (asíntota dashed gray).

%\PassOptionsToPackage{gray}{xcolor}

\documentclass[border=5mm]{standalone}
\input{../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[font=\small, >=Latex]
\begin{axis}[
    width=13cm, height=5.5cm,
    axis lines=middle,
    axis line style={->, thin},
    xmin=-0.3, xmax=9.6,
    ymin=-0.2, ymax=3.6,
    xtick=\empty,
    ytick={0.4, 2},
    yticklabels={$y_0$, $y_\infty$},
    yticklabel style={font=\footnotesize},
    xlabel={$t$},
    xlabel style={at={(axis description cs:1.0, 0.085)}, anchor=west, font=\footnotesize},
    ylabel={$y(t)$},
    ylabel style={font=\footnotesize, rotate=0, anchor=south east,
                  at={(axis description cs:0, 1)}},
    clip=false,
    samples=200,
    axis on top,
]
    % --- 2. Fills / regiones: banda del transitorio ---
    \addplot[draw=none, fill=cyan!15] coordinates
        {(0,-0.2) (0,3.5) (4,3.5) (4,-0.2)} \closedcycle;

    % --- 3. Curvas principales ---
    % Asíntota: nivel permanente y_infty
    \addplot[gray, very thin, dashed, domain=0:9.4] {2};
    % Respuesta: y(t) = y_infty - (y_infty - y_0) e^{-sigma t} cos(omega t)
    \addplot[utnblue, semithick, domain=0:9.4] {2 - 1.6*exp(-1.1*x)*cos(deg(1.8*x))};

    % --- 4. Anotaciones ---
    % Estado inicial
    \node[circle, fill=utnblue, inner sep=0pt, minimum size=3.2pt] at (axis cs:0,0.4){};
    % Etiquetas de los dos regímenes
    \node[anchor=south, font=\footnotesize] at (axis cs:2,3.18) {transitorio};
    \node[anchor=north, font=\footnotesize, gray!50!black] at (axis cs:7.4,1.9) {permanente};
\end{axis}
\end{tikzpicture}
\end{document}
