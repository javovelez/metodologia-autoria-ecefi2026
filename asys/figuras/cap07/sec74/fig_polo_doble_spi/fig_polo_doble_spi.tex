\documentclass[border=5mm]{standalone}
\input{../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

% --- Plano s ---
\draw[->, thin] (-2.0, 0) -- (1.3, 0) node[below right, font=\footnotesize] {$\sigma$};
\draw[->, thin] (0, -1.7) -- (0, 1.9) node[above, font=\footnotesize] {$j\omega$};
\fill[black] (0, 0) circle (0.8pt);
\node[red, font=\large] at (-1.3, 0) {$\times$};
\node[red, above right, font=\scriptsize, inner sep=1pt] at (-1.22, 0.10) {$(2)$};
\draw (-1.3, 0.08) -- (-1.3, -0.08);
\node[below, font=\footnotesize] at (-1.3, -0.06) {$-a$};

% --- Conector ---
\draw[->, gray, thin] (1.8, 0) -- (3.0, 0)
    node[midway, above, font=\footnotesize, gray] {$\mathcal{L}^{-1}$};

% --- h(t) ---
\begin{scope}[shift={(3.5, 0)}]
    \draw[->, thin] (-0.3, 0) -- (4.0, 0) node[right, font=\footnotesize] {$t$};
    \draw[->, thin] (0, -0.5) -- (0, 1.9) node[above, font=\footnotesize] {$h(t)$};
    % t e^{-a t} normalizado: pico 1.5 en t = 1.2 (a = 0.833).
    \draw[red!80!black, thick, smooth, samples=120, domain=0:3.8]
        plot (\x, {3.4*\x*exp(-0.833*\x)});
\end{scope}

\end{tikzpicture}
