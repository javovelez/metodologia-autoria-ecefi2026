\begin{document}

\begin{tikzpicture}[>=Latex, font=\small,
    block/.style={draw=utnblue!60, line width=0.5pt, fill=utnblue!8,
                  minimum width=2.1cm, minimum height=1cm, rounded corners=2pt,
                  blur shadow={shadow blur steps=4, shadow blur extra rounding=0pt,
                               shadow xshift=0.4pt, shadow yshift=-0.6pt, shadow opacity=25}},
    sum/.style={draw=utnblue!60, line width=0.5pt, circle,
                minimum size=0.9cm, inner sep=0pt, fill=white,
                blur shadow={shadow blur steps=4, shadow blur extra rounding=0pt,
                             shadow xshift=0.3pt, shadow yshift=-0.4pt, shadow opacity=20}},
    cross_line/.style={utnblue!40, line width=0.4pt},
    signal/.style={->, thick, utnblue},
    wire/.style={-, thick, utnblue},
    label_style/.style={font=\footnotesize, above},
]
    \node[sum]   (S) at (-3.2,0)  {};
    \node[block] (G) at (0,0)     {$G(s)$};
    \node[block] (F) at (0,-2.5)  {$F(s)$};

    % Cruz interna del sumador: entrada O (+) y entrada S (-)
    \draw[cross_line] ([xshift=-0.318cm,yshift=-0.318cm]S.center) -- ([xshift=0.318cm,yshift=0.318cm]S.center);
    \draw[cross_line] ([xshift=-0.318cm,yshift=0.318cm]S.center)  -- ([xshift=0.318cm,yshift=-0.318cm]S.center);
    \node[font=\scriptsize] at ([xshift=-0.27cm]S.center) {$+$};
    \node[font=\scriptsize] at ([yshift=-0.27cm]S.center) {$-$};

    % Camino directo
    \draw[signal] (-5.0,0)  -- node[label_style]{$X(s)$} (S.west);
    \draw[signal] (S.east)  -- node[label_style]{$E(s)$} (G.west);

    % Salida con ramificación (T-junction) hacia el lazo
    \coordinate (tap) at (2.3,0);
    \draw[wire]    (G.east) -- (tap);
    \fill[utnblue] (tap) circle (2.2pt);
    \draw[signal]  (tap) -- node[label_style]{$Y(s)$} (4.2,0);

    % Camino de realimentación (derecha a izquierda)
    \draw[signal] (tap)    |- (F.east);
    \draw[signal] (F.west) -| node[font=\footnotesize, above, pos=0.25]{$F(s)\,Y(s)$} (S.south);
\end{tikzpicture}
