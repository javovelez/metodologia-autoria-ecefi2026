\documentclass[border=5mm]{standalone}
\input{../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

% --- Plano s ---
\draw[->, thin] (-2.0, 0) -- (1.3, 0) node[below right, font=\footnotesize] {$\sigma$};
\draw[->, thin] (0, -1.9) -- (0, 2.0) node[above, font=\footnotesize] {$j\omega$};
\fill[black] (0, 0) circle (0.8pt);
\node[red, font=\large] at (-0.8, 1.3) {$\times$};
\node[red, font=\large] at (-0.8, -1.3) {$\times$};

% --- Conector ---
\draw[->, gray, thin] (1.8, 0) -- (3.0, 0)
    node[midway, above, font=\footnotesize, gray] {$\mathcal{L}^{-1}$};

% --- h(t) ---
\begin{scope}[shift={(3.5, 0)}]
    \draw[->, thin] (-0.3, 0) -- (4.0, 0) node[right, font=\footnotesize] {$t$};
    \draw[->, thin] (0, -1.7) -- (0, 1.9) node[above, font=\footnotesize] {$h(t)$};
    % Envolvente exponencial
    \draw[gray, thin, dashed, smooth, samples=60, domain=0:3.8]
        plot (\x, {1.5*exp(-0.5*\x)});
    \draw[gray, thin, dashed, smooth, samples=60, domain=0:3.8]
        plot (\x, {-1.5*exp(-0.5*\x)});
    % Sinusoide amortiguada
    \draw[red!80!black, thick, smooth, samples=220, domain=0:3.8]
        plot (\x, {1.5*exp(-0.5*\x)*cos(deg(3*\x))});
    \fill[red!80!black] (0, 1.5) circle (1.1pt);
\end{scope}

\end{tikzpicture}
\end{document}
