% ============================================================
% fig_roc_signo_a.tex
% ROC de la exponencial causal e^{at} u(t) según el signo de a,
% con los casos concretos del Ejemplo 1 de §7.1.3:
%   (a) a = -2 (señal decreciente): ROC Re{s} > -2, eje jw DENTRO  => hay TFTC
%   (b) a =  2 (señal creciente):   ROC Re{s} >  2, eje jw FUERA   => no hay TFTC
% Estilo 2D consistente con fig_roc_tres_formas (mismo capítulo).
% ============================================================
%\PassOptionsToPackage{gray}{xcolor}

\documentclass[border=5mm]{standalone}
\input{../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=0.85, font=\small]

% El polo se ubica en sigma = -1.5 / +1.5 (coordenadas) y se rotula
% con los valores concretos a = -2 / a = 2 usados en la prosa.

% =====================================================================
% Panel (a): a = -2 — polo a la izquierda del eje imaginario.
%   La ROC Re{s} > -2 cruza el eje imaginario => sí hay TFTC.
% =====================================================================
\begin{scope}[shift={(0,0)}]
    % --- 2. Fills ---
    \fill[cyan!15] (-1.5, -2.5) rectangle (3, 2.5);

    % --- 3. Ejes y fronteras ---
    \draw[->, thick] (-3, 0) -- (3.2, 0) node[right] {$\sigma$};
    \draw[->, thick] (0, -2.5) -- (0, 2.7) node[above right, inner sep=1pt] {$j\omega$};
    \draw[dashed, thick, gray] (-1.5, -2.5) -- (-1.5, 2.5);

    % --- 4. Anotaciones ---
    \fill[black] (0,0) circle (1pt);

    \node[red, font=\normalsize] at (-1.5, 0) {$\times$};
    \node[red, above left, font=\scriptsize, inner sep=1pt]
        at (-1.5, 0.05) {$s\!=\!-2$};

    \draw (-1.5, 0.08) -- (-1.5, -0.08);
    \node[below, font=\scriptsize] at (-1.5, -0.1) {$-2$};

    \node[utnblue!80!black, font=\scriptsize] at (2.3, 1.7) {ROC};

    \node[anchor=south, font=\small] at (0, 2.9) {(a)};
\end{scope}

% =====================================================================
% Panel (b): a = 2 — polo a la derecha del eje imaginario.
%   La ROC Re{s} > 2 deja al eje imaginario afuera => no hay TFTC.
% =====================================================================
\begin{scope}[shift={(7,0)}]
    \fill[cyan!15] (1.5, -2.5) rectangle (3, 2.5);

    \draw[->, thick] (-3, 0) -- (3.2, 0) node[right] {$\sigma$};
    \draw[->, thick] (0, -2.5) -- (0, 2.7) node[above right, inner sep=1pt] {$j\omega$};
    \draw[dashed, thick, gray] (1.5, -2.5) -- (1.5, 2.5);

    \fill[black] (0,0) circle (1pt);

    \node[red, font=\normalsize] at (1.5, 0) {$\times$};
    \node[red, above left, font=\scriptsize, inner sep=1pt]
        at (1.5, 0.05) {$s\!=\!2$};

    \draw (1.5, 0.08) -- (1.5, -0.08);
    \node[below, font=\scriptsize] at (1.5, -0.1) {$2$};

    \node[utnblue!80!black, font=\scriptsize] at (2.45, 1.7) {ROC};

    \node[anchor=south, font=\small] at (0, 2.9) {(b)};
\end{scope}

\end{tikzpicture}
\end{document}
