% fig_tren_pulsos_limite
% Tres paneles apilados verticalmente con tau fijo y T_0 creciente.
% Orientacion apilada: la alineacion temporal de t = 0 en los tres
% paneles hace inmediata la lectura de como, manteniendo el pulso
% central inalterado, los pulsos vecinos se separan al crecer T_0.
%
% Convencion de pulso: outline (izq + arriba + der), sin sombreado,
% para coherencia con fig_par_pulso_sinc.
%
% Puntos suspensivos a izquierda y derecha de cada panel indican que
% la senal es periodica y continua mas alla de la ventana mostrada.
%
% Figura ejemplificativa (sin valores numericos sobre los ejes): los
% valores tau = 1.5 y T_0 = 6, 10, 14 estan elegidos para que el pulso
% se lea como rectangulo (no como stem), para que cada panel muestre
% pocos periodos (5/3/3), y para que en el panel (b) el proximo pulso
% caiga claramente fuera del rango (en +-20, con xmax = 17).
%
% Sizing: 3 paneles apilados, width 9cm, height 3cm por panel
% (aspect 3:1, mas compacto en altura que el preset S4).

\begin{document}
\begin{tikzpicture}[>={Latex[length=3pt,width=3pt]}, font=\scriptsize]

\begin{groupplot}[
    group style={
        group size=1 by 3,
        vertical sep=0.9cm,
        x descriptions at=edge bottom,
    },
    width=9cm,
    height=3cm,
    xmin=-17, xmax=17,
    xtick={0},
    every tick label/.style={font=\footnotesize},
    xlabel={$t$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.18)}},
    axis lines=center,
    axis line style={->, thin},
    ymin=-0.45, ymax=1.30,
    ytick={0},
    enlarge x limits=0.02,
    clip mode=individual,
    title style={font=\scriptsize, yshift=-2pt},
]
