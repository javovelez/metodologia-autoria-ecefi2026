% fig_rc_sobre_pulso_aperiodico
% Cuatro paneles 2x2.
% Parametros: a = 1, tau = 2.
% (a) |X(jw)| = tau * |sinc(w*tau/2)| = 2 * |sin(w)/w|.
% (b) |H(jw)| = 1/sqrt(a^2 + w^2) = 1/sqrt(1 + w^2).
% (c) |Y(jw)| = |X * H| = (2 * |sin(w)/w|) / sqrt(1 + w^2).
% (d) y(t): cero para t < -1, carga (1 - exp(-(t+1))) para -1 <= t <= 1,
%     descarga exp(-(t-1)) * (1 - exp(-2)) para t > 1.
% Sizing: 4 paneles 2x2, w 7cm, h 3.5cm.

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

% ===== Panel (a): |X(jw)| =====
\begin{axis}[
    name=panel_a,
    width=7cm, height=3.5cm,
    axis lines=center,
    axis line style={->, thick},
    xmin=-15, xmax=15, ymin=-0.3, ymax=2.4,
    xtick={-12.566, -6.283, 0, 6.283, 12.566},
    xticklabels={$-4\pi$,$-2\pi$,$0$,$2\pi$,$4\pi$},
    ytick={0},
    every tick label/.style={font=\footnotesize},
    xlabel={$\omega$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.13)}},
    ylabel={$|X(j\omega)|$},
    ylabel style={font=\small, rotate=0, anchor=south east,
                  at={(axis description cs:0,1)}, xshift=-2pt},
    title={(a)},
    title style={font=\small, yshift=-2pt},
    enlarge x limits=0.02,
    clip mode=individual,
]
    \addplot[utnblue, very thick, domain=-15:15, samples=400, samples y=0]
        ({x}, {abs(2*sin(deg(x))/x)});
    \node[anchor=south west, font=\footnotesize] at (axis cs:0.4, 2.02) {$2$};
\end{axis}

% ===== Panel (b): |H(jw)| =====
\begin{axis}[
    name=panel_b,
    at={($(panel_a.east)+(1.6cm,0)$)}, anchor=west,
    width=7cm, height=3.5cm,
    axis lines=center,
    axis line style={->, thick},
    xmin=-15, xmax=15, ymin=-0.15, ymax=1.2,
    xtick={-12.566, -6.283, 0, 6.283, 12.566},
    xticklabels={$-4\pi$,$-2\pi$,$0$,$2\pi$,$4\pi$},
    ytick={0},
    every tick label/.style={font=\footnotesize},
    xlabel={$\omega$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.16)}},
    ylabel={$|H(j\omega)|$},
    ylabel style={font=\small, rotate=0, anchor=south east,
                  at={(axis description cs:0,1)}, xshift=-2pt},
    title={(b)},
    title style={font=\small, yshift=-2pt},
    enlarge x limits=0.02,
    clip mode=individual,
]
    \addplot[utnblue, very thick, domain=-15:15, samples=300]
        {1/sqrt(1 + x*x)};
    \node[anchor=south west, font=\footnotesize] at (axis cs:0.4, 1.0) {$1$};
\end{axis}

% ===== Panel (c): |Y(jw)| =====
\begin{axis}[
    name=panel_c,
    at={($(panel_a.south)+(0,-1.4cm)$)}, anchor=north,
    width=7cm, height=3.5cm,
    axis lines=center,
    axis line style={->, thick},
    xmin=-15, xmax=15, ymin=-0.3, ymax=2.4,
    xtick={-12.566, -6.283, 0, 6.283, 12.566},
    xticklabels={$-4\pi$,$-2\pi$,$0$,$2\pi$,$4\pi$},
    ytick={0},
    every tick label/.style={font=\footnotesize},
    xlabel={$\omega$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.13)}},
    ylabel={$|Y(j\omega)|$},
    ylabel style={font=\small, rotate=0, anchor=south east,
                  at={(axis description cs:0,1)}, xshift=-2pt},
    title={(c)},
    title style={font=\small, yshift=-2pt},
    enlarge x limits=0.02,
    clip mode=individual,
]
    \addplot[utnblue, very thick, domain=-15:15, samples=400]
        ({x}, {abs(2*sin(deg(x))/x)/sqrt(1 + x*x)});
    \node[anchor=south west, font=\footnotesize] at (axis cs:0.4, 2.02) {$2$};
\end{axis}

% ===== Panel (d): y(t) =====
\begin{axis}[
    name=panel_d,
    at={($(panel_c.east)+(1.6cm,0)$)}, anchor=west,
    width=7cm, height=3.5cm,
    axis lines=center,
    axis line style={->, thick},
    xmin=-2, xmax=6, ymin=-0.15, ymax=1.1,
    xtick={-1, 0, 1, 3, 5},
    xticklabels={$-\tau/2$, $0$, $\tau/2$, , },
    ytick={0, 0.865},
    yticklabels={$0$, },
    every tick label/.style={font=\footnotesize},
    xlabel={$t$},
    xlabel style={anchor=west, at={(axis description cs:1.0,0.13)}},
    ylabel={$y(t)$},
    ylabel style={font=\small, rotate=0, anchor=south east,
                  at={(axis description cs:0,1)}, xshift=-2pt},
    title={(d)},
    title style={font=\small, yshift=-2pt},
    enlarge x limits=0.02,
    clip mode=individual,
]
    % t < -1: y = 0
    \addplot[utnblue, very thick, domain=-2:-1, samples=2] {0};
    % -1 <= t <= 1: y = 1 - exp(-(t+1))
    \addplot[utnblue, very thick, domain=-1:1, samples=80] {1 - exp(-(x+1))};
    % t > 1: y = exp(-(t-1)) * (1 - exp(-2))
    \addplot[utnblue, very thick, domain=1:6, samples=80] {(1 - exp(-2)) * exp(-(x-1))};
    % Pulso de entrada como referencia ligera
    \addplot[gray, dashed, thick] coordinates {(-2,0) (-1,0) (-1,1) (1,1) (1,0) (6,0)};
\end{axis}

\end{tikzpicture}
\end{document}
