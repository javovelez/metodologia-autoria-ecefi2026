% y(t) = Π_T(t-T/2) * e^{-at}u(t)  con T=2, a=1
%
%   Zona I  (t<0):     y=0
%   Zona II (0≤t≤2):   y = 1 - e^{-t}
%   Zona III(t>2):     y = (e^2-1)·e^{-t}

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

\pgfmathsetmacro{\T}{2}
\pgfmathsetmacro{\val}{exp(2)-1}   % ≈ 6.389, valor máximo en t=T

\begin{axis}[
    width=8cm, height=5cm,
    axis lines=center,
    xlabel={$t$},
    ylabel={$y(t)$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    ylabel style={font=\small, rotate=0, anchor=south east, at={(axis description cs:0,1)}},
    xmin=-1.2, xmax=6.5,
    ymin=-0.1, ymax=1.1,
    clip=false,
    axis line style={->, thick},
    xtick={0,2},
    xticklabels={$0$, $T$},
    ytick={0},
    extra y ticks={0.865},
    extra y tick style={font=\small},
    every tick label/.style={font=\small},
]
    % Zona I: y=0 para t<0
    \addplot[domain=-1.1:0, samples=2, very thick, utnblue] {0};

    % Zona II: y = 1-e^{-t} para 0 ≤ t ≤ 2
    \addplot[domain=0:2, samples=120, very thick, utnblue] {1 - exp(-x)};

    % Zona III: y = (e^2-1)·e^{-t} para t>2
    \addplot[domain=2:6.3, samples=120, very thick, utnblue] {(exp(2)-1)*exp(-x)};

    % Continuidad en t=0 y t=T
    \fill[utnblue] (axis cs:0, 0) circle (2pt);
    \fill[utnblue] (axis cs:2, 0.865) circle (2pt);  % valor en T: 1-e^{-2}≈0.865

    % Líneas verticales punteadas en t=0 y t=T
    \draw[dashed, gray!60] (axis cs:0, -0.08) -- (axis cs:0, 0.25);
    \draw[dashed, gray!60] (axis cs:2, -0.08) -- (axis cs:2, 0.865);

    % Etiquetas de zonas
    \node[font=\footnotesize, gray, above] at (axis cs:-0.6, 0)    {Zona~I};
    \node[font=\footnotesize, gray]        at (axis cs:1, -0.09)   {Zona~II};
    \node[font=\footnotesize, gray]        at (axis cs:4.2, -0.09) {Zona~III};

    % % Flechas indicando cada tramo
    % \draw[->, thin, gray!70] (axis cs:-0.6, -0.04) -- (axis cs:-0.2, 0);
    % \draw[->, thin, gray!70] (axis cs:1, -0.065) -- (axis cs:1, 0.3);
    % \draw[->, thin, gray!70] (axis cs:4.2, -0.065) -- (axis cs:4.0, 0.12);

    % Ecuaciones de cada zona dentro del gráfico
    \node[font=\footnotesize, utnblue, align=left] at (axis cs:0.95, 0.85)
        {$\dfrac{1-e^{-at}}{a}$};
    \node[font=\footnotesize, utnblue, align=left] at (axis cs:4.2, 0.35)
        {$\dfrac{e^{aT}-1}{a}\,e^{-at}$};

\end{axis}
\end{tikzpicture}
