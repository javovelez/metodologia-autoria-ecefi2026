% y(t) = pulso_causal_a(t) * pulso_causal_b(t)  con a=1, b=2  (a ≤ b)
%
% Resultado: trapecio con
%   Zona I  (t < 0):           y = 0
%   Zona II (0 ≤ t ≤ a=1):     y = t
%   Zona III(a ≤ t ≤ b=2):     y = a = 1            (meseta)
%   Zona IV (b ≤ t ≤ a+b=3):   y = a+b - t = 3 - t
%   Zona V  (t > a+b=3):       y = 0

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

\begin{axis}[
    width=11cm, height=6cm,
    axis lines=center,
    xlabel={$t$},
    ylabel={$y(t)$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    ylabel style={font=\small, rotate=0, anchor=south east, at={(axis description cs:0,1)}},
    xmin=-0.8, xmax=4.2,
    ymin=-0.15, ymax=1.25,
    clip=false,
    axis line style={->, thick},
    xtick={0,1,2,3},
    xticklabels={$0$, $a$, $b$, $a+b$},
    ytick={0,1},
    yticklabels={$0$, $a$},
    every tick label/.style={font=\footnotesize},
    grid=major, grid style={gray!12},
    axis on top,
]
    % Relleno del trapecio
    \fill[utnblue!15]
        (axis cs:0,0) -- (axis cs:1,1) -- (axis cs:2,1) -- (axis cs:3,0) -- cycle;

    % Contorno del trapecio (sólido azul)
    \addplot[very thick, utnblue] coordinates {
        (-0.6, 0) (0, 0) (1, 1) (2, 1) (3, 0) (4.0, 0)
    };

    % Puntos de quiebre
    \fill[utnblue] (axis cs:0, 0) circle (2.2pt);
    \fill[utnblue] (axis cs:1, 1) circle (2.2pt);
    \fill[utnblue] (axis cs:2, 1) circle (2.2pt);
    \fill[utnblue] (axis cs:3, 0) circle (2.2pt);

    % Dimensiones del trapecio
    % Ancho total a+b=3
    \draw[<->, gray!70, thin] (axis cs:0,-0.13) -- (axis cs:3,-0.13);
    \node[font=\tiny, gray] at (axis cs:1.5,-0.19) {$a+b$};

    % Meseta b-a=1
    \draw[<->, gray!70, thin] (axis cs:1, 1.12) -- (axis cs:2, 1.12);
    \node[font=\tiny, gray, above] at (axis cs:1.5, 1.12) {$b-a$};

    % Flancos de duración a=1
    \draw[<->, gray!70, thin] (axis cs:0, 0.65) -- (axis cs:1, 0.65);
    \node[font=\tiny, gray, left] at (axis cs:0, 0.65) {$a$};
    \draw[<->, gray!70, thin] (axis cs:2, 0.65) -- (axis cs:3, 0.65);
    \node[font=\tiny, gray, right] at (axis cs:3, 0.65) {$a$};

    % Etiqueta del caso especial a=b
    \node[font=\tiny, gray!80, align=center] at (axis cs:3.7, 0.8) {$(a{=}b)$\\triángulo};

\end{axis}
\end{tikzpicture}
\end{document}
