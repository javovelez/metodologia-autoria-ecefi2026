\input{../../../../setup.tex}
% \PassOptionsToPackage{gray}{xcolor}

% Convolución de una rampa con un impulso desplazado.
%   x(τ)         = (τ/a)·[u(τ) - u(τ-a)],  a   = 1
%   δ(τ - t_0)   con t_0 = 2
%   y(t)         = x(t - t_0): rampa trasladada al intervalo [t_0, t_0+a]
%
% Tres paneles horizontales con rangos de t alineados:
%   (a) x(τ)         — rampa de soporte acotado [0, a]
%   (b) δ(τ - t_0)   — impulso desplazado al instante t_0
%   (c) y(t)         — rampa trasladada al instante t_0

\begin{document}
\begin{tikzpicture}[>=Latex, font=\small]

\colorlet{xcol}{utnblue}
\colorlet{hcol}{red!80!black}

\pgfmathsetmacro{\aval}{1}
\pgfmathsetmacro{\tcero}{2}
\pgfmathsetmacro{\tfin}{\tcero + \aval}

\pgfmathsetmacro{\xmn}{-1.4}
\pgfmathsetmacro{\xmx}{4.4}
\pgfmathsetmacro{\ymn}{-0.22}
\pgfmathsetmacro{\ymx}{1.45}

% =====================================================================
% PANEL (a): x(τ) — rampa de soporte acotado
% =====================================================================
\begin{axis}[
    name=panA,
    width=6.6cm, height=4.6cm,
    axis lines=center,
    xlabel={$\tau$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    xmin=\xmn, xmax=\xmx,
    ymin=\ymn, ymax=\ymx,
    clip=true,
    axis line style={->, thick},
    xtick={0, \aval},
    xticklabels={$0$, $a$},
    ytick={0, 1},
    every tick label/.style={font=\tiny},
    title={\footnotesize (a)\; $x(\tau)$},
]
    \fill[xcol!18]
        (axis cs:0,0) -- (axis cs:\aval,1) -- (axis cs:\aval,0) -- cycle;

    \addplot[xcol, very thick, domain=\xmn:0,    samples=2] {0};
    \addplot[xcol, very thick, domain=0:\aval,   samples=2] {x/\aval};
    \draw[xcol, very thick] (axis cs:\aval,1) -- (axis cs:\aval,0);
    \addplot[xcol, very thick, domain=\aval:\xmx, samples=2] {0};

    \fill[xcol]  (axis cs:0,0) circle (1.6pt);
    \fill[xcol]  (axis cs:\aval,1) circle (1.6pt);
    \fill[white] (axis cs:\aval,0) circle (1.6pt);
    \draw[xcol]  (axis cs:\aval,0) circle (1.6pt);
\end{axis}

% =====================================================================
% PANEL (b): δ(τ - t_0)
% =====================================================================
\begin{axis}[
    name=panB,
    at={(panA.east)}, anchor=west, xshift=0.6cm,
    width=6.6cm, height=4.6cm,
    axis lines=center,
    xlabel={$\tau$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    xmin=\xmn, xmax=\xmx,
    ymin=\ymn, ymax=\ymx,
    clip=false,
    axis line style={->, thick},
    xtick={0, \tcero},
    xticklabels={$0$, $t_0$},
    ytick={1},
    every tick label/.style={font=\tiny},
    title={\footnotesize (b)\; $\delta(\tau - t_0)$},
]
    \draw[->, hcol, very thick, line width=1.4pt]
        (axis cs:\tcero, 0) -- (axis cs:\tcero, 1);
\end{axis}

% =====================================================================
% PANEL (c): y(t) = x(t - t_0)
% =====================================================================
\begin{axis}[
    name=panC,
    at={(panB.east)}, anchor=west, xshift=0.6cm,
    width=6.6cm, height=4.6cm,
    axis lines=center,
    xlabel={$t$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    xmin=\xmn, xmax=\xmx,
    ymin=\ymn, ymax=\ymx,
    clip=true,
    axis line style={->, thick},
    xtick={0, \tcero, \tfin},
    xticklabels={$0$, $t_0$, $t_0{+}a$},
    ytick={0, 1},
    every tick label/.style={font=\tiny},
    title={\footnotesize (c)\; $y(t) = x(t - t_0)$},
]
    \fill[xcol!18]
        (axis cs:\tcero,0) -- (axis cs:\tfin,1) -- (axis cs:\tfin,0) -- cycle;

    \addplot[xcol, very thick, domain=\xmn:\tcero,   samples=2] {0};
    \addplot[xcol, very thick, domain=\tcero:\tfin,  samples=2]
        {(x - \tcero)/\aval};
    \draw[xcol, very thick] (axis cs:\tfin,1) -- (axis cs:\tfin,0);
    \addplot[xcol, very thick, domain=\tfin:\xmx,    samples=2] {0};

    \fill[xcol]  (axis cs:\tcero,0) circle (1.6pt);
    \fill[xcol]  (axis cs:\tfin,1) circle (1.6pt);
    \fill[white] (axis cs:\tfin,0) circle (1.6pt);
    \draw[xcol]  (axis cs:\tfin,0) circle (1.6pt);

    % Marca de la traslación de bloque por t_0
    \draw[<->, gray!70, thin]
        (axis cs:0, 0.30) --
        node[below, font=\tiny, gray]{$t_0$}
        (axis cs:\tcero, 0.30);
\end{axis}

\end{tikzpicture}
\end{document}
