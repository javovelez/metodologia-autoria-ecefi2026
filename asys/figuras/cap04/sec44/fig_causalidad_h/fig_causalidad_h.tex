\begin{tikzpicture}[>=Latex, font=\small]

% ---- Panel (a): h(t) = e^{-t}u(t)  [causal] ----
\begin{axis}[
    name=ax1,
    width=7cm, height=5cm,
    axis lines=center,
    xlabel={$t$},
    ylabel={$h(t)$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    ylabel style={font=\small, rotate=0, anchor=south east, at={(axis description cs:0,1)}},
    xmin=-2.5, xmax=4.5,
    ymin=-0.15, ymax=1.3,
    clip=false,
    axis line style={->, thick},
    xtick={-2,-1,0,1,2,3,4},
    ytick={0,1},
    every tick label/.style={font=\footnotesize},
    title={\small (a) Causal:\; $h(t)=e^{-t}u(t)$},
]
    % zero for t < 0
    \addplot[domain=-2.3:0, samples=2, very thick, utnblue] {0};
    % exponential for t >= 0
    \addplot[domain=0:4.3, samples=150, very thick, utnblue] {exp(-x)};
    % vertical jump at t=0
    \draw[utnblue, thick] (axis cs:0,0) -- (axis cs:0,1);
    \fill[utnblue] (axis cs:0,1) circle (2.5pt);
    \fill[white]   (axis cs:0,0) circle (2.5pt);
    \draw[utnblue, thick] (axis cs:0,0) circle (2.5pt);
\end{axis}

% ---- Panel (b): h(t) = e^{-|t|}  [no causal] ----
\begin{axis}[
    name=ax2,
    at={(ax1.east)}, anchor=west, xshift=1.8cm,
    width=7cm, height=5cm,
    axis lines=center,
    xlabel={$t$},
    ylabel={$h(t)$},
    xlabel style={anchor=north west, at={(axis description cs:1,0)}},
    ylabel style={font=\small, rotate=0, anchor=south east, at={(axis description cs:0,1)}},
    xmin=-4.5, xmax=4.5,
    ymin=-0.15, ymax=1.3,
    clip=false,
    axis line style={->, thick},
    xtick={-4,-3,-2,-1,0,1,2,3,4},
    ytick={0,1},
    every tick label/.style={font=\footnotesize},
    title={\small (b) No causal:\; $h(t)=e^{-|t|}$},
]
    % e^t for t < 0
    \addplot[domain=-4.3:0, samples=150, very thick, utnblue] {exp(x)};
    % e^{-t} for t >= 0
    \addplot[domain=0:4.3, samples=150, very thick, utnblue] {exp(-x)};
    \fill[utnblue] (axis cs:0,1) circle (2.5pt);
\end{axis}

\end{tikzpicture}
