% |z| <= 1  AND  0 <= Arg{z} <= pi/3
% Sector circular cerrado 0° a 60° del disco unitario
\documentclass[border=5mm]{standalone}
\input{../../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=2.8, font=\small]

    % 1. Grilla fina
    \draw[help lines, gray!30, step=0.5] (-0.25,-0.25) grid (1.25,1.10);

    % 2. Región: sector 0° a 60°
    \fill[cyan!15] (0,0) -- (1,0) arc(0:60:1) -- cycle;

    % 3. Ejes
    \draw[->, thick] (-0.20,0) -- (1.35,0) node[right] {$\operatorname{Re}$};
    \draw[->, thick] (0,-0.20) -- (0,1.20) node[above] {$\operatorname{Im}$};
    \node[below left, font=\scriptsize] at (0,0) {$0$};

    % 4. Bordes incluidos
    \draw[utnblue, thick] (0,0) -- (1,0);
    \draw[utnblue, thick] (0,0) -- ({cos(60)},{sin(60)});
    \draw[utnblue, thick] (1,0) arc(0:60:1);

    % 5. Marcas y etiqueta de ángulo
    \foreach \x in {0.5,1.0}
        \draw (\x,0.035)--(\x,-0.035) node[below, font=\scriptsize]{$\x$};
    \foreach \y in {0.5,1.0}
        \draw (0.035,\y)--(-0.035,\y) node[left, font=\scriptsize]{$\y$};
    \draw[gray, thin] (0.27,0) arc(0:60:0.27);
    \node[gray, font=\scriptsize] at (0.35,0.17) {$\pi/3$};
    \fill (0,0) circle (0.8pt);

\end{tikzpicture}
\end{document}
