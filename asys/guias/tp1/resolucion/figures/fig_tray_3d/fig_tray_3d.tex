\documentclass[border=5mm]{standalone}
\input{../../../../../setup.tex}

\begin{document}
    \begin{tikzpicture}[>=Latex, font=\small]

        % Configuración común para los 4 paneles
        \pgfplotsset{
            width=4.8cm, height=4.8cm,
            scale only axis,
            view={55}{30},
            axis lines=center,
            xlabel={$t$},
            ylabel={$\operatorname{Re}$},
            zlabel={$\operatorname{Im}$},
            xlabel style={anchor=north east, inner sep=1pt},
            ylabel style={anchor=north west, inner sep=1pt},
            zlabel style={anchor=south, inner sep=1pt},
            xmin=-0.3, xmax=3.3,
            ticks=none,
            axis line style={->},
            clip=false
        }

        % =============================================
        % PANEL (a): s = -1   (σ=-1, ω=0)
        % Trayectoria: decaimiento real sobre el eje Re
        % f(t) = e^{-t},  Im{f}=0
        % =============================================
        \begin{axis}[
            at={(0cm,0cm)},
            name=plot1,
            ymin=-1.2, ymax=1.2,
            zmin=-0.6, zmax=0.6
        ]
            \addplot3[domain=0:3, samples=120, samples y=0, thick, orange!80!black]
                ({x}, {exp(-x)}, {0});
        \end{axis}
        \node[anchor=north, yshift=6mm] at (plot1.south) {\textbf{(a) $s=-1$}};

        % =============================================
        % PANEL (b): s = j   (σ=0, ω=2π)
        % Trayectoria: hélice de radio constante (sin crecimiento)
        % f(t) = e^{jωt},  |f|=1
        % =============================================
        \begin{axis}[
            at={(5.3cm,0cm)},
            name=plot2,
            ymin=-1.3, ymax=1.3,
            zmin=-1.3, zmax=1.3
        ]
            \addplot3[domain=0:3, samples=350, samples y=0, thick, utnblue]
                ({x}, {cos(360*x)}, {sin(360*x)});
        \end{axis}
        \node[anchor=north, yshift=6mm] at (plot2.south) {\textbf{(b) $s=j$}};

        % =============================================
        % PANEL (c): s = -1+j   (σ=-1, ω=2π)
        % Trayectoria: hélice convergente (espiral que decae)
        % f(t) = e^{-t}·e^{jωt}
        % =============================================
        \begin{axis}[
            at={(0cm,-6.0cm)},
            name=plot3,
            ymin=-1.2, ymax=1.2,
            zmin=-1.2, zmax=1.2
        ]
            \addplot3[domain=0:3, samples=350, samples y=0, thick, green!50!black]
                ({x}, {exp(-x)*cos(360*x)}, {exp(-x)*sin(360*x)});
        \end{axis}
        \node[anchor=north, yshift=6mm] at (plot3.south) {\textbf{(c) $s=-1+j$}};

        % =============================================
        % PANEL (d): s = 0.5+j   (σ=0.5, ω=2π)
        % Trayectoria: hélice divergente (espiral que crece)
        % f(t) = e^{0.5t}·e^{jωt}
        % =============================================
        \begin{axis}[
            at={(5.3cm,-6.0cm)},
            name=plot4,
            ymin=-5, ymax=5,
            zmin=-5, zmax=5
        ]
            \addplot3[domain=0:3, samples=350, samples y=0, thick, red!70!black]
                ({x}, {exp(0.5*x)*cos(360*x)}, {exp(0.5*x)*sin(360*x)});
        \end{axis}
        \node[anchor=north, yshift=6mm] at (plot4.south) {\textbf{(d) $s=0.5+j$}};

    \end{tikzpicture}
\end{document}
