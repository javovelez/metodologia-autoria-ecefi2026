% 0 < Arg{z} < 2*pi/3  —  sector angular abierto (0° a 120°)
% Ray a 120°: dirección (cos120°, sin120°) = (-0.5, 0.866)
% → el view debe extenderse a x negativos para mostrarla correctamente.
% Intersección del rayo con el borde izquierdo (x=-1.8): r = 1.8/0.5 = 3.6
\documentclass[border=5mm]{standalone}
\input{../../../../../setup.tex}

\begin{document}
\begin{tikzpicture}[>=Latex, scale=0.78, font=\small]

    \useasboundingbox (-2.0,-0.6) rectangle (3.3,3.3);

    % 1. Grilla (clipeada a la vista)
    \begin{scope}
        \clip (-2.0,-0.6) rectangle (3.3,3.3);
        \draw[help lines, gray!30, step=1] (-2.0,-0.6) grid (3.3,3.3);
    \end{scope}

    % 2. Región: sector 0° a 120° (clipeado a la vista)
    \begin{scope}
        \clip (-2.0,-0.6) rectangle (3.3,3.3);
        \fill[cyan!15] (0,0) -- (4.5,0) arc(0:120:4.5) -- cycle;
    \end{scope}

    % 3. Ejes (encima del relleno)
    \draw[->, thick] (-2.1,0) -- (3.5,0) node[right] {$\operatorname{Re}$};
    \draw[->, thick] (0,-0.5) -- (0,3.5) node[above] {$\operatorname{Im}$};
    \node[below right, font=\scriptsize] at (0,0) {$0$};

    % 4. Bordes excluidos: semirrectas a 0° y a 120°, ambas clipeadas a la vista
    \begin{scope}
        \clip (-2.0,-0.6) rectangle (3.3,3.3);
        \draw[utnblue, thick, dashed, ->] (0,0) -- (3.5, 0);
        \draw[utnblue, thick, dashed, ->] (0,0) -- ({4.0*cos(120)},{4.0*sin(120)});
    \end{scope}

    % Origen excluido
    \draw[utnblue, fill=white, thick] (0,0) circle (2.5pt);

    % 5. Marcas y etiqueta de ángulo
    \foreach \x in {-1,1,2,3}
        \draw (\x,0.07)--(\x,-0.07) node[below, font=\scriptsize]{$\x$};
    \foreach \y in {1,2,3}
        \draw (0.07,\y)--(-0.07,\y) node[left, font=\scriptsize]{$\y$};
    \draw[gray, thin] (0.7,0) arc(0:120:0.7);
    \node[gray, font=\scriptsize] at ({0.52*cos(60)},{0.52*sin(60)+0.08}) {$\frac{2\pi}{3}$};

\end{tikzpicture}
\end{document}
