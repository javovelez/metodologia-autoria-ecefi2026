\documentclass[tikz, border=3pt]{standalone}

\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta}
\usepackage{xcolor}

\definecolor{utnblue}{RGB}{0, 51, 102}

\pgfplotsset{
    secuencia/.style={
        axis lines=middle,
        width=9cm, height=3.4cm,
        xmin=-1.6, xmax=6.6,
        ymin=-2.8, ymax=5.6,
        xtick={-1,1,2,3,4,5,6},
        ytick={-2,2,4},
        tick label style={font=\tiny},
        label style={font=\scriptsize},
        title style={font=\scriptsize, at={(axis description cs:1,1)},
                     anchor=north east, xshift=-2pt, yshift=-2pt},
        xticklabel style={fill=white, inner sep=1.5pt},
        every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west},
        every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south},
        enlargelimits=false,
        clip mode=individual,
    },
}

\begin{document}
\begin{tikzpicture}[font=\scriptsize]

\begin{groupplot}[
    group style={group size=1 by 3, vertical sep=1.05cm},
    secuencia,
]

% ---------------- (a) x[n] ----------------
\nextgroupplot[title={(a)}, ylabel={$x[n]$}, xlabel={$n$}]
\addplot[ycomb, utnblue, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnblue, draw=utnblue}]
    coordinates {(0,1) (1,2) (2,-1)};
\addplot[only marks, utnblue, mark=*, mark size=1.3pt]
    coordinates {(-1,0) (3,0) (4,0) (5,0) (6,0)};

% ---------------- (b) h[n] ----------------
\nextgroupplot[title={(b)}, ylabel={$h[n]$}, xlabel={$n$}]
\addplot[ycomb, red!80!black, semithick, mark=*, mark size=1.3pt,
         mark options={fill=red!80!black, draw=red!80!black}]
    coordinates {(0,2) (1,1) (3,-1)};
\addplot[only marks, red!80!black, mark=*, mark size=1.3pt]
    coordinates {(-1,0) (2,0) (4,0) (5,0) (6,0)};

% ---------------- (c) y[n] ----------------
\nextgroupplot[title={(c)}, ylabel={$y[n]$}, xlabel={$n$}]
\addplot[ycomb, red!80!black, semithick, mark=*, mark size=1.3pt,
         mark options={fill=red!80!black, draw=red!80!black}]
    coordinates {(0,2) (1,5) (3,-2) (4,-2) (5,1)};
\addplot[only marks, red!80!black, mark=*, mark size=1.3pt]
    coordinates {(-1,0) (2,0) (6,0)};

\end{groupplot}

\end{tikzpicture}
\end{document}
