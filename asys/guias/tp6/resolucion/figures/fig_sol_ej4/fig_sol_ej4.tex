\documentclass[tikz, border=3pt]{standalone}

\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta}
\usepackage{xcolor}

\definecolor{utnblue}{RGB}{0, 51, 102}
\definecolor{utnred}{RGB}{204, 0, 0}

\pgfplotsset{
    secuencia/.style={
        axis lines=middle,
        width=8cm, height=3.8cm,
        xmin=-3.6, xmax=3.6,
        ymin=-4.2, ymax=6.8,
        xtick={-3,-2,-1,1,2,3},
        ytick={-3,3,6},
        tick label style={font=\tiny},
        label style={font=\scriptsize},
        title style={font=\scriptsize, at={(axis description cs:1,1)},
                     anchor=north east, xshift=-2pt, yshift=-2pt},
        xticklabel style={fill=white, inner sep=1.5pt},
        every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west},
        every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south},
        enlargelimits=false,
        clip mode=individual,
    },
}

\begin{document}
\begin{tikzpicture}[font=\scriptsize]

\begin{groupplot}[
    group style={group size=1 by 3, vertical sep=1.2cm},
    secuencia,
]

% ---------------- (a) x[n] ----------------
\nextgroupplot[title={(a)}, ylabel={$x[n]$}, xlabel={$n$}]
\addplot[ycomb, utnblue, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnblue, draw=utnblue}]
    coordinates {(-2,6) (-1,4) (0,-2) (1,-2)};
\addplot[only marks, utnblue, mark=*, mark size=1.3pt]
    coordinates {(-3,0) (2,0) (3,0)};

% ---------------- (b) x_p[n] ----------------
\nextgroupplot[title={(b)}, ylabel={$x_p[n]$}, xlabel={$n$}]
\addplot[ycomb, utnred, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnred, draw=utnred}]
    coordinates {(-2,3) (-1,1) (0,-2) (1,1) (2,3)};
\addplot[only marks, utnred, mark=*, mark size=1.3pt]
    coordinates {(-3,0) (3,0)};

% ---------------- (c) x_i[n] ----------------
\nextgroupplot[title={(c)}, ylabel={$x_i[n]$}, xlabel={$n$}]
\addplot[ycomb, utnred, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnred, draw=utnred}]
    coordinates {(-2,3) (-1,3) (1,-3) (2,-3)};
\addplot[only marks, utnred, mark=*, mark size=1.3pt]
    coordinates {(-3,0) (0,0) (3,0)};

\end{groupplot}

\end{tikzpicture}
\end{document}
