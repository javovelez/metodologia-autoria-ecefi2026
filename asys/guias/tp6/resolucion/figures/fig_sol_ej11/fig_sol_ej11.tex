\documentclass[tikz, border=3pt]{standalone}

\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta}
\usepackage{xcolor}

\definecolor{utnblue}{RGB}{0, 51, 102}

\pgfplotsset{
    secuencia/.style={
        axis lines=middle,
        width=6.6cm, height=4cm,
        xmin=-4.6, xmax=7.6,
        ymin=-1.8, ymax=4.8,
        xtick={-4,-2,2,4,6},
        ytick={-1,1,2,3,4},
        tick label style={font=\tiny},
        label style={font=\scriptsize},
        title style={font=\scriptsize, at={(axis description cs:1,1)},
                     anchor=north east, xshift=-2pt, yshift=-2pt},
        xticklabel style={fill=white, inner sep=1.5pt},
        every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west},
        every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south},
        enlargelimits=false,
        clip mode=individual,
        xlabel={$n$},
    },
    muestras/.style={
        ycomb, utnblue, semithick, mark=*, mark size=1.2pt,
        mark options={fill=utnblue, draw=utnblue},
    },
    ceros/.style={only marks, utnblue, mark=*, mark size=1.2pt},
}

\begin{document}
\begin{tikzpicture}[font=\scriptsize]

\begin{groupplot}[
    group style={group size=2 by 2, horizontal sep=1.7cm, vertical sep=1.3cm},
    secuencia,
]

% ---------------- (a) x1 ----------------
\nextgroupplot[title={(a)}, ylabel={$x_1[n]$}]
\addplot[muestras] coordinates {(-2,1) (-1,1) (0,1) (1,1) (2,1)};
\addplot[ceros] coordinates {(-4,0) (-3,0) (3,0) (4,0) (5,0) (6,0) (7,0)};

% ---------------- (b) x2 ----------------
\nextgroupplot[title={(b)}, ylabel={$x_2[n]$}]
\addplot[muestras] coordinates {(-1,1) (0,1) (1,2) (2,2) (3,1) (4,1)};
\addplot[ceros] coordinates {(-4,0) (-3,0) (-2,0) (5,0) (6,0) (7,0)};

% ---------------- (c) x3 ----------------
\nextgroupplot[title={(c)}, ylabel={$x_3[n]$}]
\addplot[muestras] coordinates {(1,1) (2,2) (3,3) (4,4)};
\addplot[ceros] coordinates {(-4,0) (-3,0) (-2,0) (-1,0) (0,0) (5,0) (6,0) (7,0)};

% ---------------- (d) x4 ----------------
\nextgroupplot[title={(d)}, ylabel={$x_4[n]$}]
\addplot[muestras] coordinates {(-1,2) (0,-1) (2,3)};
\addplot[ceros] coordinates {(-4,0) (-3,0) (-2,0) (1,0) (3,0) (4,0) (5,0) (6,0) (7,0)};

\end{groupplot}

\end{tikzpicture}
\end{document}
