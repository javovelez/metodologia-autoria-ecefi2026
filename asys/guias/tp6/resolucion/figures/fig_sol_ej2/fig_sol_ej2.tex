\documentclass[tikz, border=3pt]{standalone}

\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usetikzlibrary{arrows.meta}
\usepackage{xcolor}

\definecolor{utnblue}{RGB}{0, 51, 102}
\definecolor{utnred}{RGB}{204, 0, 0}

% Estilo comun de los paneles de secuencia
\pgfplotsset{
    secuencia/.style={
        axis lines=middle,
        width=8cm, height=3.6cm,
        xmin=-4.6, xmax=6.6,
        ymin=-3.1, ymax=3.9,
        xtick={-4,-3,-2,-1,1,2,3,4,5,6},
        ytick={-2,2},
        tick label style={font=\tiny},
        label style={font=\scriptsize},
        title style={font=\scriptsize, at={(axis description cs:1,1)},
                     anchor=north east, xshift=-2pt, yshift=-2pt},
        xticklabel style={fill=white, inner sep=1.5pt},
        every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west},
        every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south},
        enlargelimits=false,
        clip mode=individual,
    },
}

\begin{document}
\begin{tikzpicture}[font=\scriptsize]

\begin{groupplot}[
    group style={group size=1 by 4, vertical sep=1.15cm},
    secuencia,
]

% ---------------- (a) x[n] ----------------
\nextgroupplot[title={(a)}, ylabel={$x[n]$}, xlabel={$n$}]
\addplot[ycomb, utnblue, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnblue, draw=utnblue}]
    coordinates {(-2,-1) (-1,2) (0,3) (1,1) (2,-2)};
\addplot[only marks, utnblue, mark=*, mark size=1.3pt]
    coordinates {(-4,0) (-3,0) (3,0) (4,0) (5,0) (6,0)};

% ---------------- (b) x[n-3] ----------------
\nextgroupplot[title={(b)}, ylabel={$x[n-3]$}, xlabel={$n$}]
\addplot[ycomb, utnred, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnred, draw=utnred}]
    coordinates {(1,-1) (2,2) (3,3) (4,1) (5,-2)};
\addplot[only marks, utnred, mark=*, mark size=1.3pt]
    coordinates {(-4,0) (-3,0) (-2,0) (-1,0) (0,0) (6,0)};

% ---------------- (c) x[-n] ----------------
\nextgroupplot[title={(c)}, ylabel={$x[-n]$}, xlabel={$n$}]
\addplot[ycomb, utnred, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnred, draw=utnred}]
    coordinates {(-2,-2) (-1,1) (0,3) (1,2) (2,-1)};
\addplot[only marks, utnred, mark=*, mark size=1.3pt]
    coordinates {(-4,0) (-3,0) (3,0) (4,0) (5,0) (6,0)};

% ---------------- (d) x[2-n] ----------------
\nextgroupplot[title={(d)}, ylabel={$x[2-n]$}, xlabel={$n$}]
\addplot[ycomb, utnred, semithick, mark=*, mark size=1.3pt,
         mark options={fill=utnred, draw=utnred}]
    coordinates {(0,-2) (1,1) (2,3) (3,2) (4,-1)};
\addplot[only marks, utnred, mark=*, mark size=1.3pt]
    coordinates {(-4,0) (-3,0) (-2,0) (-1,0) (5,0) (6,0)};

\end{groupplot}

\end{tikzpicture}
\end{document}
