\usepackage{xcolor}

\definecolor{utnblue}{RGB}{0, 51, 102}
\definecolor{utnred}{RGB}{204, 0, 0}

\begin{document}
\begin{tikzpicture}[
    impulso/.style={utnblue, line width=0.8pt, -{Latex[length=3pt,width=2.6pt]}},
  ]

\begin{groupplot}[
    group style={group size=3 by 2, horizontal sep=1.5cm, vertical sep=1.6cm},
    width=5.6cm, height=3.5cm,
    axis lines=middle,
    enlargelimits=false,
    tick label style={font=\scriptsize},
    title style={font=\footnotesize},
    every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west},
]

% (a) x=1
\nextgroupplot[title={(a)\ $x(t)=1$}, xmin=-4,xmax=4,ymin=0,ymax=1.3,
    xtick={0},xticklabels={$0$},ytick={1},yticklabels={$2\pi$}]

% (b) e^{jw0(t-2)}
\nextgroupplot[title={(b)\ $x(t)=e^{j\omega_0(t-2)}$}, xmin=-4,xmax=4,ymin=0,ymax=1.3,
    xtick={2},xticklabels={$\omega_0$},ytick=\empty]
\draw[impulso] (axis cs:2,0) -- (axis cs:2,1);

% (c) sin(w0(5-t))  -- magnitud de los pesos
\nextgroupplot[title={(c)\ $x(t)=\sin(\omega_0(5-t))$}, xmin=-4,xmax=4,ymin=0,ymax=1.3,

\draw[impulso] (axis cs:-2,0) -- (axis cs:-2,0.5);
\draw[impulso] (axis cs: 2,0) -- (axis cs: 2,0.5);

% (d) derivada de escalones -> 2cos(2w) (continuo)
\nextgroupplot[title={(d)\ $\frac{d}{dt}\{u(t{+}2)+u(t{-}2)\}$}, xmin=-4,xmax=4,ymin=-2.4,ymax=2.4,
    xtick=\empty, ytick={-2,2}, yticklabels={$-2$,$2$}, xlabel={$\omega$}]
\addplot[utnred, thick, samples=200, domain=-4:4] {2*cos(deg(2*x))};

% (e) 1 + cos(6 pi t + pi/8)
\nextgroupplot[title={(e)\ $1+\cos(6\pi t+\pi/8)$}, xmin=-22,xmax=22,ymin=0,ymax=1.3,
    xtick={-18.85,0,18.85},xticklabels={$-6\pi$,$0$,$6\pi$},ytick=\empty]
\draw[impulso] (axis cs:0,0) -- (axis cs:0,1);
\draw[impulso] (axis cs:-18.85,0) -- (axis cs:-18.85,0.5);
\draw[impulso] (axis cs: 18.85,0) -- (axis cs: 18.85,0.5);

\end{groupplot}
